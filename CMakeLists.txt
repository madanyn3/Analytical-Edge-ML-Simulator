cmake_minimum_required(VERSION 3.22)
project(edge_workload LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGE_SANITIZE "Enable undefined/address sanitizers" OFF)
add_library(edge_fc src/fc.cpp src/memory.cpp src/ini.cpp src/config.cpp src/service.cpp src/hierarchy.cpp src/testbench.cpp)
target_include_directories(edge_fc PUBLIC include)
target_compile_options(edge_fc PRIVATE -Wall -Wextra -Wconversion -Werror)
if(EDGE_SANITIZE)
  target_compile_options(edge_fc PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edge_fc PUBLIC -fsanitize=address,undefined)
endif()
add_executable(fc_example examples/fc_example.cpp)
target_link_libraries(fc_example PRIVATE edge_fc)
enable_testing()
add_executable(fc_tests tests/fc_tests.cpp)
target_link_libraries(fc_tests PRIVATE edge_fc)
add_test(NAME fc_contract COMMAND fc_tests)
add_executable(memory_example examples/memory_example.cpp)
target_link_libraries(memory_example PRIVATE edge_fc)
add_executable(memory_tests tests/memory_tests.cpp)
target_link_libraries(memory_tests PRIVATE edge_fc)
add_test(NAME memory_contract COMMAND memory_tests)
add_executable(edge_bench examples/edge_bench.cpp)
target_link_libraries(edge_bench PRIVATE edge_fc)
add_executable(hierarchy_tests tests/hierarchy_tests.cpp)
target_link_libraries(hierarchy_tests PRIVATE edge_fc)
add_test(NAME hierarchy_contract COMMAND hierarchy_tests)
add_test(NAME bench_hierarchy COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/hierarchy.ini)
add_test(NAME bench_latency COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/memory-latency.ini)
add_test(NAME bench_bandwidth COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/memory-bandwidth.ini)
add_test(NAME bench_csv COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/hierarchy-directed.ini
    --trace ${CMAKE_CURRENT_SOURCE_DIR}/configs/reads.csv)
