cmake_minimum_required(VERSION 3.22)
project(edge_workload LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGE_SANITIZE "Enable undefined/address sanitizers" OFF)
add_library(edge_fc src/fc.cpp src/memory.cpp src/fc_cache.cpp src/ini.cpp src/config.cpp src/cluster_config.cpp src/noc.cpp src/network.cpp src/cluster.cpp src/cluster_report.cpp src/cluster_analytic.cpp src/service.cpp src/hierarchy.cpp src/testbench.cpp)
target_include_directories(edge_fc PUBLIC include)
target_compile_options(edge_fc PRIVATE -Wall -Wextra -Wconversion -Werror)
if(EDGE_SANITIZE)
  target_compile_options(edge_fc PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(edge_fc PUBLIC -fsanitize=address,undefined)
endif()
add_executable(fc_example examples/fc_example.cpp)
target_link_libraries(fc_example PRIVATE edge_fc)
enable_testing()
add_executable(fc_tests tests/fc_tests.cpp)
target_link_libraries(fc_tests PRIVATE edge_fc)
add_test(NAME fc_contract COMMAND fc_tests)
add_executable(memory_example examples/memory_example.cpp)
target_link_libraries(memory_example PRIVATE edge_fc)
add_executable(memory_tests tests/memory_tests.cpp)
target_link_libraries(memory_tests PRIVATE edge_fc)
add_test(NAME memory_contract COMMAND memory_tests)
add_executable(fc_cache_tests tests/fc_cache_tests.cpp)
target_link_libraries(fc_cache_tests PRIVATE edge_fc)
add_test(NAME fc_cache_theorem COMMAND fc_cache_tests)
find_package(Threads REQUIRED)
add_executable(fc_cache_campaign tests/fc_cache_campaign.cpp)
target_link_libraries(fc_cache_campaign PRIVATE edge_fc Threads::Threads)
add_test(NAME fc_cache_campaign_quick COMMAND fc_cache_campaign --quick)
add_executable(edge_bench examples/edge_bench.cpp)
target_link_libraries(edge_bench PRIVATE edge_fc)
add_executable(hierarchy_tests tests/hierarchy_tests.cpp)
target_link_libraries(hierarchy_tests PRIVATE edge_fc)
add_test(NAME hierarchy_contract COMMAND hierarchy_tests)
add_test(NAME bench_hierarchy COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/hierarchy.ini)
add_test(NAME bench_latency COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/memory-latency.ini)
add_test(NAME bench_bandwidth COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/memory-bandwidth.ini)
add_test(NAME bench_csv COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/hierarchy-directed.ini
    --trace ${CMAKE_CURRENT_SOURCE_DIR}/configs/reads.csv)
add_executable(cluster_config_tests tests/cluster_config_tests.cpp)
target_link_libraries(cluster_config_tests PRIVATE edge_fc)
add_test(NAME cluster_config COMMAND cluster_config_tests ${CMAKE_CURRENT_SOURCE_DIR}/configs/cluster-2x2-mesh.ini
    ${CMAKE_CURRENT_SOURCE_DIR}/configs/hierarchy.ini)
add_executable(noc_tests tests/noc_tests.cpp)
target_link_libraries(noc_tests PRIVATE edge_fc)
add_test(NAME noc_geometry COMMAND noc_tests)
add_executable(network_tests tests/network_tests.cpp)
target_link_libraries(network_tests PRIVATE edge_fc)
add_test(NAME network_contention COMMAND network_tests)
add_executable(cluster_tests tests/cluster_tests.cpp)
target_link_libraries(cluster_tests PRIVATE edge_fc)
add_test(NAME cluster_evaluator COMMAND cluster_tests)
add_executable(cluster_theorem_tests tests/cluster_theorem_tests.cpp)
target_link_libraries(cluster_theorem_tests PRIVATE edge_fc)
add_test(NAME cluster_theorems COMMAND cluster_theorem_tests)
foreach(cluster 2x2-mesh 3x3-torus 4x4-mesh)
    add_test(NAME bench_cluster_${cluster} COMMAND edge_bench ${CMAKE_CURRENT_SOURCE_DIR}/configs/cluster-${cluster}.ini)
endforeach()
